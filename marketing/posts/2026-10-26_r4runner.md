Regenerate the numbers right before posting (they move every 4 hours):

```bash
python3 - <<'PY'
import json, collections, datetime
d=json.load(open('docs/data/inventory_live.json')); f={n:i for i,n in enumerate(d['fields'])}
yf={n:i for i,n in enumerate(d['yardFields'])}; Y=d['yards']
ex=json.load(open('docs/data/vehicle_extras.json'))['extras']
cut=(datetime.date.today()-datetime.timedelta(days=7)).isoformat()
gen=lambda y: '3rd' if 1996<=y<=2002 else '2nd' if 1990<=y<=1995 else '4th' if 2003<=y<=2009 else 'other'
o=collections.defaultdict(lambda:{'n':0,'new':0,'st':collections.Counter(),'v8':0})
for r in d['vehicles']:
    if r[f['make']]!='Toyota' or str(r[f['model']]).replace(' ','').lower()!='4runner': continue
    g=gen(int(r[f['year']] or 0)); o[g]['n']+=1
    if str(r[f['dateAdded']] or '')[:10]>=cut: o[g]['new']+=1
    o[g]['st'][Y[r[f['yard']]][yf['state']]]+=1
    x=ex.get(str(r[f['id']])); 
    if x and '8-cyl' in str(x[2] if len(x)>2 else ''): o[g]['v8']+=1
for g in ('2nd','3rd','4th'): print(g, o[g]['n'], 'new7d', o[g]['new'], o[g]['st'].most_common(3), 'v8', o[g]['v8'])
PY
```
